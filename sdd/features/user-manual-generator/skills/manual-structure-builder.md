# Skill: manual-structure-builder

**Skill ID:** manual-structure-builder
**Feature:** ANCLORA-UMG-001
**Version:** v1.0
**Type:** Content Generation
**Runtime:** Python 3.11+

---

## 1. Purpose

Build a hierarchical, well-structured user manual from the functionality map generated by `manual-content-analyzer`. This skill transforms raw functionality data into human-readable documentation with proper organization, flow, and clarity.

---

## 2. Inputs

| Input | Type | Source | Required |
|-------|------|--------|----------|
| Functionality Map | JSON | Output from `manual-content-analyzer` | Yes |
| Manual Template | Markdown | `templates/manual-template.md` | Optional |
| Content Rules | Markdown | `rules/user-manual-rules.md` | Yes |
| i18n Language | String | `es` (default), `en`, `de`, `ru` | No |

---

## 4. Processing Logic

### 4.1 Structure Building Algorithm

```python
from typing import Dict, List
import json

class ManualStructureBuilder:
    def __init__(self, functionality_map: Dict, language: str = "es"):
        self.func_map = functionality_map
        self.language = language
        self.manual = {"metadata": {}, "sections": []}

    def build(self) -> Dict:
        """Build complete manual structure"""
        self.add_metadata()
        self.add_introduction()
        self.add_navigation_section()
        self.add_feature_sections()
        self.add_use_cases()
        self.add_troubleshooting()
        return self.manual

    def add_metadata(self):
        """Add manual metadata"""
        self.manual["metadata"] = {
            "title": "Manual de Usuario: Anclora Nexus",
            "version": self.extract_app_version(),
            "date": datetime.now().strftime("%Y-%m-%d"),
            "language": self.language,
            "status": "current"
        }

    def add_introduction(self):
        """Add introduction section"""
        section = {
            "level": 1,
            "title": "Introducción",
            "content": self.generate_intro_content(),
            "subsections": [
                {
                    "level": 2,
                    "title": "Qué es Anclora Nexus",
                    "content": self.generate_what_is_content()
                },
                {
                    "level": 2,
                    "title": "Requisitos previos",
                    "content": self.generate_requirements_content()
                },
                {
                    "level": 2,
                    "title": "Acceso a la plataforma",
                    "content": self.generate_access_content()
                }
            ]
        }
        self.manual["sections"].append(section)

    def add_navigation_section(self):
        """Add navigation documentation"""
        section = {
            "level": 1,
            "title": "Navegación Principal",
            "content": "Anclora Nexus se organiza en dos componentes principales de navegación:",
            "subsections": [
                {
                    "level": 2,
                    "title": "Sidebar (Menú Lateral)",
                    "content": self.generate_sidebar_docs()
                },
                {
                    "level": 2,
                    "title": "Header (Barra Superior)",
                    "content": self.generate_header_docs()
                }
            ]
        }
        self.manual["sections"].append(section)

    def add_feature_sections(self):
        """Add documentation for each sidebar section"""
        for section_data in self.func_map["sidebar"]["sections"]:
            section = {
                "level": 1,
                "title": f"Sección {section_data['title'].upper()}",
                "content": self.generate_section_intro(section_data),
                "subsections": []
            }

            for link in section_data["links"]:
                page_info = self.find_page_info(link["href"])
                subsection = {
                    "level": 2,
                    "title": link["name"],
                    "content": self.generate_page_docs(page_info, link)
                }
                section["subsections"].append(subsection)

            self.manual["sections"].append(section)

    def generate_sidebar_docs(self) -> str:
        """Generate sidebar documentation"""
        docs = ["## Estructura del Sidebar\n"]
        docs.append("El menú lateral se organiza en 3 secciones colapsables:\n")

        for section in self.func_map["sidebar"]["sections"]:
            docs.append(f"\n### {section['title']}\n")
            docs.append(f"**Opciones ({len(section['links'])}):**\n")
            for link in section["links"]:
                status = "✅" if link["page_exists"] else "⏳"
                docs.append(f"- {status} **{link['name']}** - `/app{link['href']}`\n")

        return "".join(docs)

    def generate_header_docs(self) -> str:
        """Generate header documentation"""
        docs = ["## Componentes del Header\n"]
        docs.append("La barra superior contiene las siguientes funcionalidades:\n\n")

        for component in self.func_map["header"]["components"]:
            docs.append(f"### {component['type']}\n")
            docs.append(f"**Función:** {component['functionality']}\n")
            docs.append(f"**Ubicación:** {component['location']}\n\n")

        return "".join(docs)

    def generate_page_docs(self, page_info: Dict, link: Dict) -> str:
        """Generate documentation for a specific page"""
        if not page_info:
            return f"⏳ Esta funcionalidad está en desarrollo.\n\n**Ruta:** `{link['href']}`\n"

        docs = []
        docs.append(f"**Ruta:** `{page_info['route']}`\n\n")

        if page_info.get("description"):
            docs.append(f"{page_info['description']}\n\n")

        if page_info.get("widgets"):
            docs.append("### Componentes Principales\n\n")
            for widget in page_info["widgets"]:
                widget_info = self.find_widget_info(widget)
                if widget_info:
                    docs.append(f"#### {widget}\n")
                    docs.append(f"{widget_info.get('description', 'Widget de visualización de datos.')}\n\n")

        if page_info.get("actions"):
            docs.append("### Acciones Disponibles\n\n")
            for action in page_info["actions"]:
                docs.append(f"- {action}\n")
            docs.append("\n")

        if page_info.get("role_access"):
            docs.append(f"**Acceso:** {', '.join(page_info['role_access'])}\n\n")

        return "".join(docs)

    def add_use_cases(self):
        """Add role-based use cases"""
        section = {
            "level": 1,
            "title": "Casos de Uso Prácticos",
            "content": "Ejemplos de uso según tu rol en la organización:",
            "subsections": [
                {
                    "level": 2,
                    "title": "Por Rol: Owner",
                    "content": self.generate_owner_use_cases()
                },
                {
                    "level": 2,
                    "title": "Por Rol: Manager",
                    "content": self.generate_manager_use_cases()
                },
                {
                    "level": 2,
                    "title": "Por Rol: Agent",
                    "content": self.generate_agent_use_cases()
                }
            ]
        }
        self.manual["sections"].append(section)

    def generate_owner_use_cases(self) -> str:
        return """### Caso: Revisión semanal de negocio

**Objetivo:** Evaluar salud comercial en 20 minutos.

**Pasos:**
1. Abre **Dashboard** y revisa **QuickStats**
2. Compara volumen de leads vs semana anterior
3. Entra a **Leads** y filtra por estado "nuevo"
4. Revisa **PropertyPipeline** para identificar cuellos de botella
5. Entra a **Team** para verificar carga de trabajo por miembro
6. Si detectas anomalías, crea tareas correctivas

**Resultado esperado:** Plan de acción claro para la semana.
"""

    def generate_manager_use_cases(self) -> str:
        return """### Caso: Gestión diaria de leads

**Objetivo:** Convertir leads en oportunidades activas.

**Pasos:**
1. Abre **Dashboard** y revisa **LeadsPulse**
2. Identifica los 5 leads de mayor prioridad
3. Abre cada lead y registra contacto (llamada/email)
4. Actualiza estado del lead según resultado
5. Crea tarea de seguimiento si es necesario
6. Vincula lead con propiedad en **Properties** si aplica

**Resultado esperado:** Leads atendidos < 24h desde entrada.
"""

    def generate_agent_use_cases(self) -> str:
        return """### Caso: Ejecución de tareas asignadas

**Objetivo:** Completar tareas del día.

**Pasos:**
1. Abre **Dashboard** y revisa **TasksToday**
2. Ordena tareas por prioridad comercial
3. Ejecuta cada tarea (llamada, email, visita)
4. Registra resultado en notas de la tarea
5. Marca tarea como completada o reprogramada
6. Si surge nueva información, actualiza el lead/propiedad asociado

**Resultado esperado:** Cero tareas vencidas al final del día.
"""

    def add_troubleshooting(self):
        """Add troubleshooting section"""
        section = {
            "level": 1,
            "title": "Troubleshooting",
            "content": "Soluciones a problemas comunes:",
            "subsections": [
                {
                    "level": 2,
                    "title": "Errores comunes",
                    "content": self.generate_common_errors()
                },
                {
                    "level": 2,
                    "title": "FAQ",
                    "content": self.generate_faq()
                }
            ]
        }
        self.manual["sections"].append(section)

    def generate_common_errors(self) -> str:
        return """### Error: "Email o contraseña incorrectos"

**Causa:** Credenciales inválidas.

**Solución:**
1. Verifica que el email sea correcto
2. Usa "Olvidé mi contraseña" para resetear
3. Si persiste, contacta al Owner para verificar tu membresía

---

### Error: "Acceso restringido"

**Causa:** Tu rol no tiene permisos para esa acción.

**Solución:**
1. Verifica tu rol en **Profile**
2. Solicita al Owner ajuste de permisos si es necesario
3. Consulta la matriz de permisos en este manual

---

### Error: "Cuenta inactiva"

**Causa:** Tu membresía no está activa en la organización.

**Solución:**
1. Contacta al Owner o Manager
2. Solicita reactivación de tu cuenta
3. Verifica que tu email esté en la lista de miembros
"""

    def generate_faq(self) -> str:
        return """### ¿Puedo usar Anclora Nexus desde móvil?

Sí, la interfaz es responsive y funciona en navegadores móviles. Para mejor experiencia, recomendamos usar tablet o desktop.

---

### ¿Cómo cambio el idioma de la interfaz?

Usa el selector de idioma (🌐) en el header superior derecho. Soportamos ES, EN, DE, RU.

---

### ¿Qué significa cada estado de lead?

- **Nuevo:** Lead recién ingresado, sin contactar
- **Contactado:** Primera interacción realizada
- **Cualificado:** Lead con potencial confirmado
- **Propuesta:** Oferta comercial enviada
- **Ganado:** Lead convertido en cliente
- **Perdido:** Lead descartado (sin interés, fuera de presupuesto, etc.)

---

### ¿Puedo exportar datos?

Sí, la mayoría de tablas tienen botón "Exportar" (CSV/Excel). Si no ves esta opción, solicita acceso al Owner.
"""

    def find_page_info(self, route: str) -> Dict:
        """Find page information by route"""
        for page in self.func_map.get("pages", []):
            if page["route"] == route:
                return page
        return None

    def find_widget_info(self, widget_name: str) -> Dict:
        """Find widget information by name"""
        for widget in self.func_map.get("widgets", []):
            if widget["name"] == widget_name:
                return widget
        return None
```

## 6. Execution Example

```python
from manual_structure_builder import ManualStructureBuilder
import json

# Load functionality map
with open("functionality-map.json", "r") as f:
    func_map = json.load(f)

# Build manual structure
builder = ManualStructureBuilder(func_map, language="es")
manual_structure = builder.build()

# Save to JSON
with open("manual-structure.json", "w") as f:
    json.dump(manual_structure, f, indent=2, ensure_ascii=False)
```
